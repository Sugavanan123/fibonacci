
n = int(input("Enter number of nodes: "))
cost = [list(map(int, input().split())) for _ in range(n)]
dist = [row[:] for row in cost]

for k in range(n):
    for i in range(n):
        for j in range(n):
            dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])

print("Shortest Distance Matrix:")
for row in dist:
    print(*row)
