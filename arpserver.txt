
import socket

arp = {
    "192.168.1.1": "AA:BB:CC:DD:EE:01",
    "192.168.1.2": "AA:BB:CC:DD:EE:02",
    "192.168.1.3": "AA:BB:CC:DD:EE:03"
}

s = socket.socket()
s.bind(("127.0.0.1", 5000))
s.listen(1)

print("Server waiting...")
c, addr = s.accept()
print("Client connected")

while True:
    msg = c.recv(1024).decode()
    if msg == "EXIT":
        break

    typ, value = msg.split()

    if typ == "ARP":
        result = "MAC: " + arp.get(value, "Not Found")
    else:
        result = "IP: " + next(
            (ip for ip, mac in arp.items() if mac == value),
            "Not Found"
        )

    c.send(result.encode())

c.close()
s.close()
