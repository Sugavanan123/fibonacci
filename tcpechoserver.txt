import socket

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

server.bind(("127.0.0.1", 12345))
server.listen(1)

print("TCP Server is running...")

conn, addr = server.accept()
print("Connected by:", addr)

data = conn.recv(1024)
print("Received from client:", data.decode())

conn.send(data)

conn.close()
server.close()
